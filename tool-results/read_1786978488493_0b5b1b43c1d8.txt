   371→  const [visible, setVisible] = useState(false)
   372→  useEffect(() => {
   373→    const el = ref.current
   374→    if (!el) return
   375→    const obs = new IntersectionObserver(
   376→      ([e]) => { if (e.isIntersecting) setVisible(true) },
   377→      { threshold }
   378→    )
   379→    obs.observe(el)
   380→    return () => obs.disconnect()
   381→  }, [threshold])
   382→  return { ref, visible }
   383→}

   385→function useCountUp(end: number, duration = 2000, start = 0) {
   386→  const [count, setCount] = useState(start)
   387→  const [running, setRunning] = useState(false)
   388→  useEffect(() => {
   389→    if (!running) return
   390→    let startTs: number | null = null
   391→    let raf: number
   392→    const step = (ts: number) => {
   393→      if (!startTs) startTs = ts
   394→      const progress = Math.min((ts - startTs) / duration, 1)
   395→      setCount(Math.floor(start + (end - start) * progress))
   396→      if (progress < 1) raf = requestAnimationFrame(step)
   397→    }
   398→    raf = requestAnimationFrame(step)
   399→    return () => cancelAnimationFrame(raf)
   400→  }, [running, end, duration, start])
   401→  return { count, start: () => setRunning(true) }
   402→}
