   179→/* ───────────────────────── SHARED HOOK ───────────────────────── */
   180→
   181→function useOnScreen(threshold = 0.15) {
   182→  const ref = useRef<HTMLDivElement>(null)
   183→  const [visible, setVisible] = useState(false)
   184→  useEffect(() => {
   185→    const el = ref.current
   186→    if (!el) return
   187→    const obs = new IntersectionObserver(
   188→      ([e]) => { if (e.isIntersecting) setVisible(true) },
   189→      { threshold }
   190→    )
   191→    obs.observe(el)
   192→    return () => obs.disconnect()
   193→  }, [threshold])
   194→  return { ref, visible }
   195→}
