   319→   319→/* ───────────────────────── SHARED HOOK ───────────────────────── */
   320→   320→
   321→   321→function useOnScreen(threshold = 0.15) {
   322→   322→  const ref = useRef<HTMLDivElement>(null)
   323→   323→  const [visible, setVisible] = useState(false)
   324→   324→  useEffect(() => {
   325→   325→    const el = ref.current
   326→   326→    if (!el) return
   327→   327→    const obs = new IntersectionObserver(
   328→   328→      ([e]) => { if (e.isIntersecting) setVisible(true) },
   329→   329→      { threshold }
   330→   330→    )
   331→   331→    obs.observe(el)
   332→   332→    return () => obs.disconnect()
   333→   333→  }, [threshold])
   334→   334→  return { ref, visible }
   335→   335→}

   736→   736→function ProjectDetailPage({ project, onClose }: { project: typeof PROJECTS[0]; onClose: () => void }) {
   737→   737→  const [currentIdx, setCurrentIdx] = useState(0)
   738→   738→  const gallery = project.gallery
   739→   739→  const scrollRef = useRef<HTMLDivElement>(null)
   740→   740→
   741→   741→  useEffect(() => {
   742→   742→    const handler = (e: KeyboardEvent) => {
   743→   743→      if (e.key === 'Escape') onClose()
   744→   744→      if (e.key === 'ArrowRight') setCurrentIdx((i) => Math.min(i + 1, gallery.length - 1))
   745→   745→      if (e.key === 'ArrowLeft') setCurrentIdx((i) => Math.max(i - 1, 0))
   746→   746→    }
   747→   747→    window.addEventListener('keydown', handler)
   748→   748→    document.body.style.overflow = 'hidden'
   749→   749→    scrollRef.current?.scrollTo(0, 0)
   750→   750→    return () => {
   751→   751→      window.removeEventListener('keydown', handler)
   752→   752→      document.body.style.overflow = ''
   753→   753→    }
   754→   754→  }, [gallery.length, onClose])
   755→   755→
   756→   756→  return (
   757→   757→    <motion.div
   758→   758→      initial={{ opacity: 0 }}
   759→   759→      animate={{ opacity: 1 }}
   760→   760→      exit={{ opacity: 0 }}
   761→   761→      transition={{ duration: 0.3 }}
   762→   762→      className="fixed inset-0 z-[100] bg-[#1A1A1A] overflow-y-auto"
   763→   763→      ref={scrollRef}
