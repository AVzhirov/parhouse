   319→/* ───────────────────────── SHARED HOOK ───────────────────────── */
   320→
   321→function useOnScreen(threshold = 0.15) {
   322→  const ref = useRef<HTMLDivElement>(null)
   323→  const [visible, setVisible] = useState(false)
   324→  useEffect(() => {
   325→    const el = ref.current
   326→    if (!el) return
   327→    const obs = new IntersectionObserver(
   328→      ([e]) => { if (e.isIntersecting) setVisible(true) },
   329→      { threshold }
   330→    )
   331→    obs.observe(el)
   332→    return () => obs.disconnect()
   333→  }, [threshold])
   334→  return { ref, visible }
   335→}

   736→function ProjectDetailPage({ project, onClose }: { project: typeof PROJECTS[0]; onClose: () => void }) {
   737→  const [currentIdx, setCurrentIdx] = useState(0)
   738→  const gallery = project.gallery
   739→  const scrollRef = useRef<HTMLDivElement>(null)
   740→
   741→  useEffect(() => {
   742→    const handler = (e: KeyboardEvent) => {
   743→      if (e.key === 'Escape') onClose()
   744→      if (e.key === 'ArrowRight') setCurrentIdx((i) => Math.min(i + 1, gallery.length - 1))
   745→      if (e.key === 'ArrowLeft') setCurrentIdx((i) => Math.max(i - 1, 0))

   747→    window.addEventListener('keydown', handler)
   748→    document.body.style.overflow = 'hidden'

   759→      animate={{ opacity: 1 }}
